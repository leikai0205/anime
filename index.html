function doGet(e) { var ss = SpreadsheetApp.getActiveSpreadsheet(); // 1. 讀取自動抓取區 var autoSheet = ss.getSheetByName("自動抓取動漫"); var autoData = parseSheetToJSON(autoSheet); // 2. 讀取手動備份區（工作表名稱為 "當季動漫"） var manualSheet = ss.getSheetByName("當季動漫"); var manualData = parseSheetToJSON(manualSheet); var result = { auto: autoData, manual: manualData }; var callback = e && e.parameter ? e.parameter.callback : null; var jsonString = JSON.stringify(result); if (callback) { var output = callback + "(" + jsonString + ")"; return ContentService.createTextOutput(output) .setMimeType(ContentService.MimeType.JAVASCRIPT); } else { return ContentService.createTextOutput(jsonString) .setMimeType(ContentService.MimeType.JSON); } } // 共用的解析函式：把工作表轉成結構化 JSON function parseSheetToJSON(sheet) { if (!sheet) return {}; var data = sheet.getDataRange().getValues(); var database = {}; var currentSeason = ""; var currentDays = null; var daysOrder = ["星期日", "星期一", "星期二", "星期三", "星期四", "星期五", "星期六"]; for (var r = 0; r < data.length; r++) { var row = data[r]; var firstCell = String(row[0]).trim(); if (firstCell.indexOf("年") !== -1 && (firstCell.indexOf("冬") !== -1 || firstCell.indexOf("春") !== -1 || firstCell.indexOf("夏") !== -1 || firstCell.indexOf("秋") !== -1)) { currentSeason = firstCell; database[currentSeason] = { "星期日": [], "星期一": [], "星期二": [], "星期三": [], "星期四": [], "星期五": [], "星期六": [] }; continue; } if (firstCell === "星期日" && currentSeason) { currentDays = []; for (var c = 0; c < 7; c++) { var dayName = String(row[c]).trim(); if (daysOrder.includes(dayName)) { currentDays.push(dayName); } } continue; } if (currentSeason && currentDays && firstCell !== "") { var hasValue = row.some(function(cell) { return cell !== ""; }); if (!hasValue) continue; var richTextValues = sheet.getRange(r + 1, 1, 1, 7).getRichTextValues()[0]; for (var c = 0; c < 7; c++) { var dayName = daysOrder[c]; var cellValue = String(row[c]).trim(); if (cellValue && !daysOrder.includes(cellValue) && cellValue !== currentSeason) { var link = ""; try { var richText = richTextValues[c]; if (richText && richText.getLinkUrl()) { link = richText.getLinkUrl(); } } catch (err) {} if (database[currentSeason] && database[currentSeason][dayName]) { database[currentSeason][dayName].push({ title: cellValue, link: link }); } } } } } return database; }
